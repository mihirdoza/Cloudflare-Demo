export default {
  async fetch(request, env) {
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        status: 204,
        headers: {
          'Access-Control-Allow-Origin': '*',
          'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
          'Access-Control-Allow-Headers': 'Content-Type, Authorization',
        },
      });
    }
    const url = new URL(request.url);

    if (request.method === 'POST') {
      
      if (url.pathname === '/bind-database') {
        return bindDatabase(request, env);
      }

      if (url.pathname === '/unbind-database') {
        return unbindDatabase(request, env);
      }
      if (url.pathname === '/manufacturer') {
        return ManufacturerSearch(request, env);
      }
      if (url.pathname === '/database/create') {
        return createDatabase(request);
      }
      if (url.pathname === '/database-import-sql') {
        const database_id = url.searchParams.get('database_id');
        return importDatabase(request, env,database_id);
      }
      if (url.pathname === '/database-upload-json') {
        return uploadJsonDatabase(request, env);
      }
      if (url.pathname === '/database-download-json') {
        return downloadJsonDatabase(request, env);
      }
      if(url.pathname==='/database-export-sql'){
        return exportDatabase(request,env)
      }

      if(url.pathname==='/backup/remove'){
        return DeleteBackup(request,env)
      }

      if(url.pathname==='/backup/download'){
        return DownloadBackup(request,env)
      }

      if(url.pathname==='/backup/initiate'){
        return DatabaseBackupInitiate(request,env)
      }
      if(url.pathname==="/purge-tables"){
        
        return RemoveAllTables(request,env);
      }

      if(url.pathname==="/backup/restore"){
        
        return BackupRestorDatabase(request,env);
      }

    }
    if (request.method === 'GET') {
      if (url.pathname === '/databases') {
        return listDatabases();
      }

      if (url.pathname === '/backup/list') {
        return BackupList(request,env);
      }

      
      

    }

    if(request.method === 'DELETE'){
        const match = url.pathname.match(/^\/database\/([^/]+)$/)

        if (match) {
          const databaseId = match[1]; // Extracted database ID
          return deleteDatabase(request,databaseId);}
      

    
    }

    return new Response('Not Found', { status: 404 });
  }
};

 const ManufacturerSearch = async (request, env) => {
  try {
    // Handle preflight CORS request (OPTIONS)
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        status: 200,
        headers: {
          'Access-Control-Allow-Origin': '*',  // Allow all origins
          'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS', // Allow methods
          'Access-Control-Allow-Headers': 'Content-Type, Authorization', // Allow headers
        }
      });
    }

    // Parse the request body
    const body = await request.json().catch(() => null);

    console.log('Request Body:', body);

    const { search, condition } = body || {};
    const queryCondition = condition || 'OR'; // Default to 'OR' if no condition is provided

    if (!search) {
      const sqlQuery = `SELECT DISTINCT Manufacturer FROM search ORDER BY Manufacturer ASC`;
      console.log('Without Search Query ==>', sqlQuery);

      try {
        const queryResult = await env.DB.prepare(sqlQuery).all();

        console.log(queryResult);
        const results = queryResult?.results || [];
        if (results.length !== 0) {
          const manufacturers = results.map((row) => row.Manufacturer);
          // Add CORS headers allowing all origins
          return new Response(JSON.stringify({ manufacturers, totalManufacturer: manufacturers.length }), {
            status: 200,
            headers: {
              'Access-Control-Allow-Origin': '*',  // Allow all origins
              'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS', // Allow methods
              'Access-Control-Allow-Headers': 'Content-Type, Authorization', // Allow headers
              'Content-Type': 'application/json'
            }
          });
        }
      } catch (err) {
        throw new Error(`${err}`);
      }
    }

    const HandleSearchWord = async (word) => {
      const Sword = await word.trim().split(/\s+/).map((word) => word.replace(/'/g, "''"));
      return Sword;
    }
    const words = await HandleSearchWord(search);

    console.log('Keywords:', words);

    const queryStrategies = [
      (word) => `${word}%`, // Query no-1: `%{word}`
      (word) => `%${word}`, // Query no-2: `{word}%`
      (word) => `%${word}%`, // Query no-3: `%{word}%`
    ];

    let results = [];
    let manufacturers = [];

    // Sequentially apply query strategies
    for (const strategy of queryStrategies) {
      // Build the SQL condition for all words using the current strategy
      const sqlConditions = words.map(
        (word) => `
        (EQID LIKE '${strategy(word)}' OR 
         MfgAcronym LIKE '${strategy(word)}' OR 
         EQType LIKE '${strategy(word)}' OR 
         MfgEQType LIKE '${strategy(word)}' OR 
         MfgProdLine LIKE '${strategy(word)}' OR 
         MfgProdNo LIKE '${strategy(word)}' OR 
         MfgDesc LIKE '${strategy(word)}' OR 
         Manufacturer LIKE '${strategy(word)}')`
      );

      // Combine conditions with the specified condition (AND/OR)
      const sqlCondition = sqlConditions.join(` ${queryCondition} `);
      const sqlQuery = `SELECT DISTINCT Manufacturer FROM search WHERE ${sqlCondition} ORDER BY Manufacturer ASC`;
      console.log('SQL Query:', sqlQuery);

      // Execute the query
      const queryResult = await env.DB.prepare(sqlQuery).all();
      results = queryResult?.results || [];

      if (results.length > 0) {
        // Break loop if results are found
        manufacturers = results.map((row) => row.Manufacturer);
        break;
      }
    }

    // Return results if any, else return empty response
    if (manufacturers.length === 0) {
      return new Response(JSON.stringify({ manufacturers: [], totalManufacturer: 0 }), {
        status: 200,
        headers: corsHeaders
      });
    }

    // Add CORS headers to the final response
    return new Response(JSON.stringify({ manufacturers, totalManufacturer: manufacturers.length }), {
      status: 200,
      headers:corsHeaders
    });
  } catch (error) {
    console.error('Error:', error.stack || error.message || error);

    // Add CORS headers in case of error
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: corsHeaders
    });
  }
};

async function bindDatabase(request, env) {

     }

console.log(existing_script)
    const formData = new FormData();

async function exportDatabase(request, env) {

    console.log("✅ Export URL Received:", exportUrl);

    console.log("🔹 Streaming SQL File...");
    const response = await fetch(exportUrl);

    if (!response.ok) {
      throw new Error('Failed to fetch SQL file');
    }

    const headers = new Headers();
    headers.set("Content-Disposition", `attachment; filename="${database_id}_export.sql"`);
    headers.set("Content-Type", "application/sql");

    return new Response(response.body, {
      status: 200,
      headers: {...headers,...corsHeaders}
    });
  } catch (error) {
    logData.push(error.message);
    console.error("Error streaming SQL file:", error.message || error);
    return new Response(JSON.stringify({
      message: error.message || "Server Error",
      logData
    }), {
      status: 500,
      headers: corsHeaders
    });
  }
}

async function importDatabase(request,env, database_id) {
  try {
    // Get file data (SQL file) from the request body
    const fileData = new Uint8Array(await request.arrayBuffer());
    if (!fileData.length) {
      return new Response(
        JSON.stringify({ error: "SQL file content is required" }),
        { status: 400, headers: { 'Content-Type': 'application/json' } }
      );
    }

    // Calculate MD5 hash of the file
    const etagBuffer = await crypto.subtle.digest('MD5', fileData);
    const etagHex = Array.from(new Uint8Array(etagBuffer))
      .map(b => b.toString(16).padStart(2, '0'))
      .join('');
    
    console.log("File etag:", etagHex);

    // Step 1: Initialize upload (action: "init")
    const initResponse = await fetch(
      `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`,
      {
        method: 'POST',
        headers,
        body: JSON.stringify({
          action: "init",
          etag: etagHex
        })
      }
    );
    const initData = await initResponse.json();
    console.log("Init Response:", initData);

    // If the file has already been processed, return success
    if (initData.result?.status === "complete") {
      return new Response(
        JSON.stringify({
          success: true,
          message: "This file was already imported",
          status: "complete",
          details: initData.result
        }),
        { status: 200, headers:corsHeaders}
      );
    }

    // If an upload_url is provided, it's a new file that needs to be uploaded
    if (initData.result?.upload_url) {
      // Step 2: Upload the SQL file using a PUT request
      const uploadResponse = await fetch(initData.result.upload_url, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/sql' },
        body: fileData
      });
      if (!uploadResponse.ok) {
        throw new Error(`Failed to upload file: ${uploadResponse.statusText}`);
      }
      console.log("File uploaded successfully");

      // Step 3: Trigger the ingest process (action: "ingest")
      let importResponse = await fetch(
        `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`,
        {
          method: 'POST',
          headers,
          body: JSON.stringify({
            action: "ingest",
            etag: etagHex,
            filename: initData.result.filename
          })
        }
      );
      let importData = await importResponse.json();
      console.log("Initial Ingest Response:", importData);

      // Poll for status if needed.
      let tries = 0;
      const maxTries = 10;
      while (tries < maxTries && importData.result?.status !== "complete") {
        await delay(2000); // wait 2 seconds before polling again

        const pollResponse = await fetch(
          `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`,
          {
            method: 'POST',
            headers,
            body: JSON.stringify({
              action: "poll",
              current_bookmark: importData.result?.at_bookmark || ""
            })
          }
        );
        importData = await pollResponse.json();
        console.log(`Poll Response (${tries + 1}):`, importData);
        if (importData.result?.status === "complete") {
          break;
        }
        tries++;
      }

      return new Response(
        JSON.stringify({ 
          success: true, 
          message: "Database imported successfully", 
          status: importData.result?.status,
          bookmark: importData.result?.at_bookmark,
          details: importData.result
        }),
        { status: 202, headers: corsHeaders }
      );
    } 
    // If there's no upload_url but a bookmark exists, then file is already in progress.
    else if (initData.result?.at_bookmark) {
      return new Response(
        JSON.stringify({
          success: true,
          message: "This file is currently being imported",
          status: "in_progress",
          bookmark: initData.result.at_bookmark,
          details: initData.result
        }),
        { status: 202, headers:corsHeaders}
      );
    }
    else {
      throw new Error("Unexpected response from import initialization");
    }

  } catch (error) {
    console.error("Import error:", error);
    return new Response(
      JSON.stringify({ 
        success: false, 
        error: error.message,
        details: error.details || null
      }),
      { status: 500, headers: corsHeaders}
    );
  }
}

// Helper function to delay for a given number of milliseconds
function delay(ms)
 {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function uploadJsonDatabase(request, env) {
 
  try {
    let database_id;
    let combinedSql = "";

    // Check if the request is multipart/form-data
    const contentType = request.headers.get("content-type") || "";
    if (contentType.includes("multipart/form-data")) {
      const formData = await request.formData();
      database_id = formData.get("database_id");
      if (!database_id) {
        return new Response(
          JSON.stringify({ error: "database_id is required as a form field" }),
          { status: 400, headers: corsHeaders }
        );
      }
      
      // Gather keys to help debug what fields are present.
      const receivedKeys = [];
      let fileFound = false;
      
      // Iterate over formData entries
      for (const [key, value] of formData.entries()) {
        receivedKeys.push(key)
;
        if (key === "database_id") continue;
        
        // Try handling both File objects (which have arrayBuffer) and plain text fields.
        let textContent = "";
        if (value && typeof value.arrayBuffer === "function") {
          textContent = await value.text();
          fileFound = true;
        } else if (typeof value === "string") {
          textContent = value;
          fileFound = true;
        }
        
        if (textContent) {
          let jsonData;
          try {
            jsonData = JSON.parse(textContent);
          } catch (err) {
            throw new Error(`Invalid JSON in field ${key}`);
          }
          combinedSql += convertJsonToSql(jsonData) + "\n";
        }
      }
      
      if (!fileFound) {
        return new Response(
          JSON.stringify({ error: "No JSON file provided in form data", formData: receivedKeys }),
          { status: 400, headers: corsHeaders }
        );
      }
    } else {
      // Fallback: If not multipart, assume JSON body with json_file or json_files.
      const body = await request.json();
      database_id = body.database_id;
      if (!database_id) {
        return new Response(
          JSON.stringify({ error: "database_id is required in JSON body" }),
          { status: 400, headers: corsHeaders }
        );
      }
      if (body.json_files && Array.isArray(body.json_files)) {
        for (const jsonData of body.json_files) {
          combinedSql += convertJsonToSql(jsonData) + "\n";
        }
      } else if (body.json_file && typeof body.json_file === "object") {
        combinedSql = convertJsonToSql(body.json_file);
      } else if (body.sql_file) {
        combinedSql = body.sql_file;
      } else {
        return new Response(
          JSON.stringify({ error: "JSON body must contain 'json_file' or 'json_files' field" }),
          { status: 400, headers: corsHeaders }
        );
      }
    }

    // Convert the combined SQL commands into a binary buffer.
    const fileData = new TextEncoder().encode(combinedSql);
    if (!fileData.length) {
      return new Response(
        JSON.stringify({ error: "Failed to generate SQL file content" }),
        { status: 400, headers: corsHeaders }
      );
    }

    // Calculate MD5 hash of the fileData.
    const hashBuffer = await crypto.subtle.digest("MD5", fileData);
    const etagHex = Array.from(new Uint8Array(hashBuffer))
      .map(b => b.toString(16).padStart(2, "0"))
      .join("");
    console.log("File etag:", etagHex);

    // Step 1: Initialize the import (action: "init")
    const initResponse = await fetch(
      `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`,
      {
        method: "POST",
        headers,
        body: JSON.stringify({
          action: "init",
          etag: etagHex
        })
      }
    );
    const initData = await initResponse.json();
    console.log("Init Response:", initData);

    // If the API reports the file is already imported, return success.
    if (initData.result?.status === "complete") {
      return new Response(
        JSON.stringify({
          success: true,
          message: "This file was already imported",
          status: "complete",
          details: initData.result
        }),
        { status: 200, headers: corsHeaders }
      );
    }

    // If an upload_url is provided, it's a new file that needs to be uploaded.
    if (initData.result?.upload_url) {
      // Step 2: Upload the SQL file via a PUT request.
      const uploadResponse = await fetch(initData.result.upload_url, {
        method: "PUT",
        headers: { "Content-Type": "application/sql" },
        body: fileData
      });
      if (!uploadResponse.ok) {
        throw new Error(`Failed to upload file: ${uploadResponse.statusText}`);
      }
      console.log("File uploaded successfully");

      // Step 3: Trigger the ingest process (action: "ingest")
      const ingestResponse = await fetch(
        `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`,
        {
          method: "POST",
          headers,
          body: JSON.stringify({
            action: "ingest",
            etag: etagHex,
            filename: initData.result.filename
          })
        }
      );
      const ingestData = await ingestResponse.json();
      console.log("Ingest Response:", ingestData);
      if (!ingestData.success) {
        throw new Error("Failed to start import");
      }

      return new Response(
        JSON.stringify({
          success: true,
          message: "Import process started successfully",
          status: "in_progress",
          bookmark: ingestData.result?.at_bookmark,
          details: ingestData.result
        }),
        { status: 202, headers:corsHeaders }
      );
    }
    // If there's a bookmark but no upload_url, then the file is already being imported.
    else if (initData.result?.at_bookmark) {
      return new Response(
        JSON.stringify({
          success: true,
          message: "This file is currently being imported",
          status: "in_progress",
          bookmark: initData.result.at_bookmark,
          details: initData.result
        }),
        { status: 202, headers: corsHeaders }
      );
    } else {
      throw new Error("Unexpected response from import initialization");
    }
  } catch (error) {
    console.error("Import error:", error);
    return new Response(
      JSON.stringify({
        success: false,
        error: error.message,
        details: error.details || null
      }),
      { status: 500, headers:corsHeaders }
    );
  }
}

// Helper function to convert a JSON object into SQL commands.
// Expects JSON format: { "tableName": [ { column: value, ... }, ... ], ... }
function convertJsonToSql(jsonData) {
  let sqlCommands = "";
  for (const [tableName, rows] of Object.entries(jsonData)) {
    if (!Array.isArray(rows)) {
      throw new Error(`Invalid JSON format: ${tableName} should be an array of rows`);
    }
    for (const row of rows) {
      const columns = Object.keys(row).join(", ");
      const values = Object.values(row)
        .map(value => {
          if (typeof value === "string") {
            return `'${value.replace(/'/g, "''")}'`;
          } else if (value === null || value === undefined) {
            return "NULL";
          } else {
            return value;
          }
        })
        .join(", ");
      sqlCommands += `INSERT OR REPLACE INTO ${tableName} (${columns}) VALUES (${values});\n`;
    }
  }
  return sqlCommands;
}

async function DatabaseBackupInitiate(request, env) {
  try {
    const { database_id, database_name } = await request.json();
    if (!database_id || !database_name) {
      return new Response(
        JSON.stringify({ success: false, error: "Missing database_id or database_name" }),
        { status: 400, headers: corsHeaders }
      );
    }

    // 1️⃣ Get Export URL
    const exportUrl = await requestExportURL1(database_id);
    console.log("🔹 Export URL Received:", exportUrl);

    // 2️⃣ Fetch SQL Dump from Cloudflare API
    const response = await fetch(exportUrl);
    if (!response.ok) {
      throw new Error("Failed to fetch SQL dump");
    }
    let sqlDump = await response.text();

    // 3️⃣ Modify SQL Dump
    sqlDump = sqlDump
      .replace(/CREATE TABLE /g, "CREATE TABLE IF NOT EXISTS ") // Ensure tables are not duplicated
      .replace(/INSERT INTO /g, "INSERT OR REPLACE INTO ") // Prevent duplicate insertions
      .replace(/CREATE INDEX /g, "CREATE INDEX IF NOT EXISTS "); // Ensure indexes are not duplicated

    // 4️⃣ Generate Backup File Name (Database Name + Timestamp)
    const timestamp = new Date().toISOString().replace(/[:.]/g, "-"); // Format: YYYY-MM-DDTHH-MM-SS
    const filename = `${database_name}_${timestamp}.sql`;

    // 5️⃣ Upload Modified SQL Dump to Cloudflare R2
    await env.MY_BUCKET.put(filename, sqlDump, {
      httpMetadata: { contentType: "text/plain" },
    });

    console.log("✅ Modified Backup saved to R2:", filename);

    return new Response(
      JSON.stringify({
        success: true,
        message: "Backup created with modified SQL",
        file: filename,
      }),
      { headers: corsHeaders}
    );
  } catch (error) {
    console.error("❌ Backup Failed:", error.message || error);
    return new Response(
      JSON.stringify({ success: false, error: error.message }),
      { status: 500, headers: corsHeaders }
    );
  }
}




/**
 * restoreBackup:
 * - Expects a JSON body with "database_id" and "backup_filename".
 * - Retrieves the backup file from R2.
 * - Computes its MD5 hash.
 * - Calls the D1 API to initialize the import.
 * - Uploads the SQL file.
 * - Triggers the ingest process.
 */
async function BackupRestorDatabase(request, env) {
  const corsHeaders = {
    "Access-Control-Allow-Origin": "*",
    "Content-Type": "application/json",
  };

  try {
    // Parse JSON body from the request.
    const { database_id, backup_filename } = await request.json();
    if (!database_id || !backup_filename) {
      return new Response(
        JSON.stringify({
          success: false,
          error: "Both 'database_id' and 'backup_filename' are required.",
        }),
        { status: 400, headers: corsHeaders }
      );
    }

    // Retrieve the backup file from R2 (bound as "MY_BUCKET" in your wrangler.toml)
    const r2Object = await env.MY_BUCKET.get(backup_filename);
    if (!r2Object) {
      return new Response(
        JSON.stringify({
          success: false,
          error: `Backup file '${backup_filename}' not found in R2.`,
        }),
        { status: 404, headers: corsHeaders }
      );
    }

    // Read file content from R2
    const fileBuffer = await r2Object.arrayBuffer();

    // Compute MD5 hash (ETag) from the file data.
    const hashBuffer = await crypto.subtle.digest("MD5", fileBuffer);
    const etagHex = Array.from(new Uint8Array(hashBuffer))
      .map((b) => b.toString(16).padStart(2, "0"))
      .join("");

    // 1. Initialize the import by calling the D1 API (action: "init")
    const initResponse = await requestUploadURL1(env, etagHex, database_id);
    if (!initResponse.result?.success) {
      throw new Error(`Upload initialization failed: ${JSON.stringify(initResponse.errors)}`);
    }

    // If the API reports the file is already processed, return success.
    if (initResponse.result.status === "complete") {
      return new Response(
        JSON.stringify({ success: true, message: "Database already imported" }),
        { headers: corsHeaders }
      );
    }

    // Extract upload URL and generated filename from the init response.
    const { upload_url, filename } = initResponse.result;

    // 2. Upload the SQL file using a PUT request (raw binary upload)
    await uploadSQLFile1(upload_url, fileBuffer);

    // 3. Trigger the ingest process (action: "ingest")
    const ingestResponse = await requestUploadURL1(env, etagHex, database_id, filename);
    if (!ingestResponse.result?.success) {
      throw new Error(`Ingest failed: ${JSON.stringify(ingestResponse.errors || ingestResponse.result?.error)}`);
    }

    return new Response(
      JSON.stringify({ success: true, message: "Database imported successfully" }),
      { headers: corsHeaders }
    );
  } catch (err) {
    return new Response(
      JSON.stringify({ success: false, error: err.message }),
      { status: 500, headers: corsHeaders }
    );
  }
}

/**
 * requestUploadURL:
 * - Calls Cloudflare's D1 API for the given database.
 * - If "filename" is not provided, it initializes the import (action "init").
 * - If "filename" is provided, it triggers the ingest (action "ingest").
 */
async function requestUploadURL1(env, etag, database_id, filename = null) {
  const url = `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/import`;
  

  // Use "init" when no filename is provided; use "ingest" when uploading is complete.
  const payload = filename
    ? { action: "ingest", etag, filename }
    : { action: "init", etag };

  const response = await fetch(url, {
    method: "POST",
    headers,
    body: JSON.stringify(payload),
  });
  return await response.json();
}

async function downloadJsonDatabase(request, env) {
  try {
    const { database_id } = await request.json();
    if (!database_id) {
      return new Response(JSON.stringify({ success: false, error: "database_id Not Found",headers:corsHeaders }), { status: 404 });
    }

    const url = `https://api.cloudflare.com/client/v4/accounts/${account_id}/d1/database/${database_id}/query`;
   

    // Get all tables
    const getTablesResponse = await fetch(url, {
      method: "POST",
      headers,
      body: JSON.stringify({ sql: "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' AND name NOT LIKE '_cf_KV' AND name NOT LIKE 'prefix%';" })
    });

    if (!getTablesResponse.ok) {
      throw new Error(`Failed to get tables: ${getTablesResponse.statusText}`);
    }

    const tablesData = await getTablesResponse.json();
    const tables = tablesData.result?.[0]?.results || [];

    if (!tables.length) {
      return new Response(JSON.stringify({ success: false, error: "No tables found in database",headers:corsHeaders }), { status: 404 });
    }

    const files = [];

    // Generate JSON for each table
    for (const table of tables) {
      const tableName = table.name;

      const tableDataResponse = await fetch(url, {
        method: "POST",
        headers,
        body: JSON.stringify({ sql: `SELECT * FROM ${tableName}` })
      });

      if (!tableDataResponse.ok) {
        throw new Error(`Failed to get data for table ${tableName}: ${tableDataResponse.statusText}`);
      }

      const tableData = await tableDataResponse.json();
      const tableJson = JSON.stringify({ [tableName]: tableData.result[0]?.results || [] }, null, 2);

      // Encode each file as base64
      const base64Content = btoa(unescape(encodeURIComponent(tableJson)));

      files.push({
        filename: `${tableName}.json`,
        content: base64Content
      });
    }

    // Return file metadata and content
    return new Response(JSON.stringify({ success: true, files }), {
      headers:corsHeaders
    });

  } catch (error) {
    console.error("Export Error:", error);
    return new Response(JSON.stringify({ success: false, error: error.message,headers:corsHeaders }), { status: 500 });
  }
}
